\section{Contributions (the claim list)}

The thesis, in one sentence, to be argued or replaced:

\begin{quote}
\emph{In a factored model the weight update is forced: any minimal realisation of a prescribed
operator step has the shared-multiplier form, and that form alone implies the balancedness
conservation law. The only remaining freedom is the \textbf{operator target}. Gradient descent's
implicit bias is therefore entirely a choice of target, and we can now make the other choice
exactly --- including at points where every first- and second-order method provably cannot move.}
\end{quote}

\begin{enumerate}[label=\textbf{C\arabic*.},leftmargin=*]
  \item \textbf{The minimal realisation problem and its certificate.} \PROVED{} \\
        Problem \eqref{eq:P}; the shared-multiplier KKT form \eqref{eq:kkt}; the residual $\rho$
        \eqref{eq:rho} as a solver-independent optimality test. \S\ref{sec:problem}.
  \item \textbf{Conservation from the KKT form alone.} \PROVED{} \MEAS{} \\
        Any flow of the form $\dot W_\ell = A_\ell^\top\Lam B_\ell^\top$ with a \emph{shared}
        $\Lam$ conserves $W_{\ell+1}^\top W_{\ell+1}-W_\ell W_\ell^\top$. The Du--Hu--Lee /
        Arora balancedness invariant is a property of minimal realisations, not of gradient
        descent. Thm~\ref{thm:conserve}. \emph{This is the one statement the survey does not
        already contain.}
  \item \textbf{Degeneracy: the escape must be constructive.} \PROVED{} \MEAS{} \\
        At a degenerate base point all derivatives of order $<L$ vanish, so every null-consistent
        derivative-based rule of order $p<L$ returns $\dW=0$ --- including natural gradient, and
        including a first-order solver applied to the \emph{exact} objective.
        Thm~\ref{thm:degen}, Cor.~\ref{cor:pinned}.
  \item \textbf{Exactness of the hierarchical recursion.} \PROVED{} \MEAS{} \\
        From a degenerate base point the recursion solves \eqref{eq:P} exactly for every $L$ iff
        each split uses the depth-proportional exponent; the symmetric split is exact iff $L$ is
        a power of two. Thm~\ref{thm:exact}. Verified to $1.000000$ for $L=2\dots128$ at $d=32$.
  \item \textbf{A closed-form cost-to-go under alignment.} \OPEN{} (partial) \\
        The two-factor split decouples into $d$ quartics when the matrices share a frame, giving a
        per-mode cost-to-go at a \emph{general} base point, $500$--$2400\times$ faster.
        Prop.~\ref{prop:aligned}. Conditional on spectral separation; gap $1.18$--$1.36$ at
        strong separation, useless at a flat spectrum.
  \item \textbf{Measurements the certificate makes possible.} \MEAS{} \\
        \S\ref{sec:measure}: natural gradient fails outside the lazy regime; the operator target
        is reachable at depth 128 so the objective was never the obstruction; the unbiased weight
        path is characterised.
  \item \textbf{The biased vs.\ unbiased trajectory study.} \OPEN{} \\
        \S\ref{sec:traj}. \TODO{this is the section that justifies the machinery and it is not
        done. Decide the setting (bottleneck) and run it.}
\end{enumerate}

\paragraph{Explicitly not claimed.} \DEAD{} A new characterisation of the implicit bias
(\cite{arora2018optimization} closed it, Thm~1); a missing closed form for fixed-gate networks
(\S\ref{sec:notclaimed}); $K_\ell\ge L$ as a hypothesis-free inequality; any universal benefit
from removing the bias.

\section{Setup}\label{sec:setup}

\begin{itemize}[leftmargin=*]
  \item \textbf{Model.} $W_\ell\in\R^{n_\ell\times n_{\ell-1}}$, $\ell=1..L$;
        $\Jop(W)=W_L\cdots W_1$ (DLN) or $W_L D_{L-1}\cdots D_1 W_1$ with fixed diagonal $D_\ell$
        (FGLN).
  \item \textbf{Contexts.} $A_\ell = W_L D_{L-1}\cdots D_\ell$, $B_\ell = D_{\ell-1}W_{\ell-1}\cdots W_1$,
        so $\Jop=A_\ell W_\ell B_\ell$ for every $\ell$. Primed $A'_\ell,B'_\ell$ denote contexts
        at the \emph{updated} weights.
  \item \textbf{Loss and operator gradient.} $\mathcal L(\Jop)=\mathbb E_x[\ell(\Jop x,y)]$,
        $G=\partial\mathcal L/\partial \Jop$. Coordinate GD: $\partial\mathcal L/\partial W_\ell = A_\ell^\top G B_\ell^\top$.
  \item \textbf{Transfer operator.} $\Tr_W(H)=\sum_\ell A_\ell A_\ell^\top H B_\ell^\top B_\ell$;
        induced step $\Delta \Jop_{\mathrm{GD}} = -\eta\,\Tr_W(G)+O(\eta^2)$.
  \item \textbf{Targets.} \emph{Biased} $-\eta\Tr_W(G)$ (what GD takes). \emph{Unbiased}
        $\Delta_\star=-\eta G$ (operator steepest descent). $\Pstar = \Jop + \Delta_\star$.
  \item \textbf{Imbalance.} $I_\ell = W_{\ell+1}^\top W_{\ell+1}-W_\ell W_\ell^\top$;
        unbalancedness magnitude $\max_\ell\|I_\ell\|_F$ (Assumption 1 of \cite{cohen2024lectures}).
\end{itemize}

\section{The minimal realisation problem}\label{sec:problem}

\begin{definition}[(P)]
\begin{equation}\label{eq:P}
  \min_{\dW}\ \sum_{\ell=1}^L \|\dW_\ell\|_F^2
  \qquad\text{s.t.}\qquad \prod_\ell (W_\ell+\dW_\ell) = \Pstar .
\end{equation}
\end{definition}

\begin{lemma}[shared multiplier]\label{lem:kkt}\PROVED{}
At a regular optimum of \eqref{eq:P} there is a single $\Lam\in\R^{n_L\times n_0}$ with
\begin{equation}\label{eq:kkt}
  \dW_\ell = A'^\top_\ell\,\Lam\,B'^\top_\ell \qquad \text{for every } \ell .
\end{equation}
\end{lemma}
\NOTE{Proof: stationarity of $\sum\|\dW_\ell\|^2-\langle\Lam,\prod(W+\dW)-\Pstar\rangle$, using
$\partial_{\dW_\ell}\langle\Lam,A'_\ell(W_\ell+\dW_\ell)B'_\ell\rangle=A'^\top_\ell\Lam B'^\top_\ell$.
App.~\ref{app:kkt}. Regularity (LICQ) is \emph{not} automatic --- see \S\ref{sec:limits}.}

\begin{definition}[certificate]
\begin{equation}\label{eq:rho}
  \rho(\dW) := \min_{\Lam}\ \frac{\sum_\ell\|\dW_\ell - A'^\top_\ell\Lam B'^\top_\ell\|_F^2}{\sum_\ell\|\dW_\ell\|_F^2}
  \ \in[0,1].
\end{equation}
\end{definition}
\NOTE{$\rho=0$ iff first-order optimal for (P); $\rho=1$ iff orthogonal to every admissible form.
Computable for \emph{any} solver's output without knowing the optimum. Matrix-free via CG on
$\Tr$; the dense map would be $1$\,GB at $d=32,L=128$.}

\paragraph{Why gradient descent is not the same thing.} \PROVED{}
GD's step $\dW_\ell=-\eta A_\ell^\top G B_\ell^\top$ has the form \eqref{eq:kkt} with
$\Lam=-2\eta G$ --- but with contexts at the \emph{current} weights, i.e. it is the minimal
realisation of the \emph{linearised} constraint $\sum_\ell A_\ell \dW_\ell B_\ell=\Delta$.
Equation \eqref{eq:kkt} is self-referential ($A',B'$ depend on $\dW$). Two consequences:
\begin{enumerate}[label=(\alph*),leftmargin=*]
  \item the bias of GD is entirely its \emph{target} $\Tr_W(G)$, not its lift;
  \item where the current contexts vanish the linearised problem is trivial and GD is pinned,
        while the exact problem remains feasible. This is Thm~\ref{thm:degen}.
\end{enumerate}

\section{Statements}\label{sec:statements}

\begin{theorem}[conservation from the KKT form]\label{thm:conserve}\PROVED{}\MEAS{}
Let $\dot W_\ell = A_\ell^\top\Lam B_\ell^\top$ for a single $\Lam$ (arbitrary, possibly
time-dependent). Then $\tfrac{d}{dt}\big(W_{\ell+1}^\top W_{\ell+1}-W_\ell W_\ell^\top\big)=0$ for
every $\ell$.
\end{theorem}
\NOTE{Proof: $\dot W_{\ell+1}^\top W_{\ell+1}=B_{\ell+1}\Lam^\top A_\ell$ and
$\dot W_\ell W_\ell^\top = A_\ell^\top \Lam B_{\ell+1}^\top$ are transposes. App.~\ref{app:conserve}.
Measured: discrepancy $1.5\times10^{-15}$ for random shared $\Lam$, versus $1.4\times10^{1}$ for an
independent per-layer step. \textbf{Consequence:} \cite{du2018algorithmic} balancedness is a
property of minimal realisations, and \cite{arora2018optimization} Thm~1's hypothesis is inherited
by \emph{both} arms, not just by GD.}

\begin{theorem}[degeneracy]\label{thm:degen}\PROVED{}
Let $\Phi(\dW)=\prod_\ell(W_\ell+\dW_\ell)$ and $F(\dW)=\|\Phi(\dW)-\Pstar\|_F^2$ with
$\Pstar\neq0$. At $W=0$, $\Phi$ is homogeneous of degree $L$, hence $D^m F(0)=0$ for
$1\le m<L$.
\end{theorem}

\begin{corollary}\label{cor:pinned}\PROVED{}\MEAS{}
Every null-consistent derivative-based rule of order $p<L$ returns $\dW=0$: GD, momentum, Adam,
Muon, Shampoo, SOAP, K-FAC, Newton, Gauss--Newton, natural gradient with pseudo-inverse Fisher
\cite{bernacchia2018exact}, and Layerwise LQR \cite{dufortlabbe2026layerwise}.
\end{corollary}
\NOTE{Measured at $d=6$, $L\in\{2,3,4\}$: min-norm reaches $\Pstar$ to $10^{-14}$ with
$\rho\sim10^{-29}$; every listed alternative returns exactly $0$. \textbf{Also}: a first-order
solver on the \emph{exact} objective is pinned too (L-BFGS on the augmented Lagrangian slides back
to the origin and is trapped for $L\ge3$), so the escape must be constructive. That is what makes
Thm~\ref{thm:exact} load-bearing rather than a convenience.}

\begin{theorem}[exactness of the recursion]\label{thm:exact}\PROVED{}\MEAS{}
From a degenerate base point, the hierarchical recursion solves \eqref{eq:P} exactly for every $L$
if and only if each split uses the depth-proportional exponent $t=a/(a+b)$. The symmetric split
$t=\tfrac12$ is exact iff every split in the recursion tree is even, i.e.\ iff $L$ is a power of
two.
\end{theorem}
\NOTE{Ingredients: the cost-to-go $V_a(Y)=a\sum_i\sigma_i(Y)^{2/a}$ (Schatten-$2/a$; the
$L$-factor form of \cite{srebro2005maximum}, general case folklore, cf.\ \cite{shang2016scalable}),
Bellman, and AM--GM for the split optimum. App.~\ref{app:exact}.}

\begin{proposition}[aligned closed form]\label{prop:aligned}\OPEN{}
If $\Pstar=U S_z V^\top$, $Y=U S_y Q^\top$, $X=Q S_x V^\top$ share a frame, the split
$\min\|Y'-Y\|^2+\|X'-X\|^2$ s.t.\ $Y'X'=Z$ separates into $d$ scalar problems with stationarity
$y^4-y_0y^3+x_0zy-z^2=0$, solvable by radicals.
\end{proposition}
\NOTE{Hypothesis is Thm~6.1 of \cite{haas2026why} (separation forces alignment), hence checkable.
Restricting to frame-diagonal factors costs exactly the off-diagonal mass, which is constant in
$y$ and so does not perturb the scalar optimisation. Gap to the reference: $1.18$--$1.36$ at
strong separation, $10^3$--$10^5$ at a flat spectrum.}

\section{Limitations, and what is not claimed}\label{sec:notclaimed}\label{sec:limits}

\begin{itemize}[leftmargin=*]
  \item \DEAD{} \textbf{Fixed-gate networks do not need new machinery.} Absorbing the gates into
        $M_\ell=D_\ell W_\ell D_{\ell-1}$ makes the FGLN flow \emph{literally} the deep linear
        flow restricted to a coordinate subspace, so \cite{arora2018optimization} Thm~1 transfers
        verbatim under $M$-balancedness, and Prop.~4.5 of \cite{haas2026why} is correct.
        An earlier claim that gates break the conservation law is withdrawn.
        The open gated object is the \emph{multi-pattern} case.
  \item \OPEN{} \textbf{The certificate needs a constraint qualification.} Lemma~\ref{lem:kkt} is
        necessary only at a \emph{regular} optimum, and the constraint Jacobian
        $M(\dW)=\sum_\ell A_\ell\dW_\ell B_\ell$ loses rank exactly where the operator collapses.
        Observed: a feasible point with smaller $\sum\|W_\ell\|^2$ but worse $\rho$.
        \TODO{characterise when LICQ holds for (P).}
  \item \textbf{Prior art to position against, not to claim.} The $2-2/L$ exponent via AM--GM is
        in \cite{rawal2026saddle}; the Riemannian metric view is \cite{bah2022learning}; removing
        depth by constraining the parameterisation is \cite{brechet2023deep}; minimal weight
        perturbation for an output change is \cite{evans2026theory}; layer-wise targets are
        \cite{frerix2018proximal}.
  \item \textbf{Deep linear + squared loss reduction.} If every step exactly realises
        $\Pstar=\Jop-\eta G$, the operator trajectory is plain GD on a convex problem. In that
        setting the method is a \emph{control}, not a speedup. State this ourselves.
\end{itemize}
