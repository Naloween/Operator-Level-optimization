\section{Contributions}\label{sec:contrib}

\NOTE{The paper is in two parts. \textbf{Part I, deep linear}, introduces the reference step and
validates it against a known closed form. \textbf{Part II, nonlinear}, carries the same objects to
ReLU and CReLU networks, where the reference step is no longer exactly realisable and the
comparison becomes substantive. Each part carries its own solver.}

\begin{enumerate}[label=\textbf{C\arabic*.},leftmargin=*,itemsep=2pt]
  \item \textbf{A reference step.} Given a prescribed operator change, the minimum-norm weight
        update that realises it: exactly for deep linear chains (S1, \S\ref{sec:solver}), and up to
        the architecture's reachable set for gated networks (S2). This supplies the thing the field
        has lacked --- a canonical answer to ``what \emph{should} this step have been'' --- against
        which any optimiser's step can be scored. \OK{}\RUN{}
  \item \textbf{A hyperparameter-free comparison protocol} (\S\ref{sec:protocol}). Every optimiser
        is scored from the \emph{same} weights against the same reference, and the directional
        scores are cosines, hence invariant to the learning rate. A comparison that does not depend
        on tuning cannot be dismissed as mis-tuned --- which is the objection that sank the previous
        submission. \RUN{}
  \item \textbf{A constructive solver for the prescribed-step problem}, exact from degenerate
        points at which every derivative-based method is stationary (\S\ref{sec:solver},
        Thms~\ref{thm:degen}--\ref{thm:exact}). \OK{}\RUN{}
  \item \textbf{Balancedness is a property of minimal realisations, not of gradient descent}
        (Thm~\ref{thm:conserve}). \OK{}\RUN{}
  \item \textbf{What the comparison finds}: trajectories and endpoints under gradient descent,
        Adam and the reference, across models, initialisations and tasks (\S\ref{sec:exp}). \TD{}
\end{enumerate}

\NOTE{\textbf{Demoted, deliberately.} The gain exponent was previously listed as a contribution. It
is a function of the current weights alone, so two optimisers evaluated at the same point give the
same exponent: it cannot discriminate between update rules, and our own measurements show it barely
discriminates between architectures either ($1.318/1.309/1.319$ for linear/CReLU/ReLU at $L{=}8$).
It is retained where it is genuinely strong --- as \emph{validation}, since reproducing
\cite{arora2018optimization} to $10^{-14}$ is the best evidence the machinery is correct, and as
\emph{motivation}, since the departure from $2-2/L$ at standard initialisation is what says there
is distortion worth correcting. The exact accounting $c_k=s_k^2R_k$ (\S\ref{sec:decomp}) stands,
but as an explanation of that departure, not as a finding about optimisers.}

\section{Notation}\label{sec:notation}

\NOTE{Every symbol used later is defined here, in order of dependence.}

\paragraph{N1. Models.} Weights $W_\ell\in\R^{n_\ell\times n_{\ell-1}}$, $\ell=1..L$.
\emph{DLN}: $P=W_L\cdots W_1$. \emph{FGLN}: $P=W_LD_{L-1}\cdots D_1W_1$ with $D_\ell$ a fixed
diagonal. \emph{ReLU MLP} (bias-free): $D_\ell(x)=\mathrm{diag}(\mathbb 1[z_\ell(x)>0])$ and
$P(x)=W_LD_{L-1}(x)\cdots D_1(x)W_1$, so $f(x)=P(x)x$ exactly. \emph{CReLU MLP}:
$D(z)=[\mathrm{diag}(\mathbb 1[z>0]);-\mathrm{diag}(\mathbb 1[z<0])]$ satisfies $D^\top D=I$, so
looks-linear init $W=[O\mid-O]$ gives $P(x)=O_L\cdots O_1$ at any depth.

\paragraph{N2. Contexts.} $B_1=I$, $B_{\ell+1}=D_\ell W_\ell B_\ell$; $A_L=I$,
$A_\ell=A_{\ell+1}W_{\ell+1}D_\ell$; so $P(x)=A_\ell(x)W_\ell B_\ell(x)$ for every $\ell$.

\paragraph{N3. Loss.} $G(x):=\partial\ell/\partial P(x)$, rank one for squared loss
($(f-y)x^\top$) and cross-entropy ($(\mathrm{softmax}(f)-e_y)x^\top$). By the chain rule
$\partial\mathcal L/\partial W_\ell=\sum_x A_\ell(x)^\top G(x)B_\ell(x)^\top$.

\paragraph{N4. Step map and target.}
$M(\dW):=\big(\sum_\ell A_\ell(x)\dW_\ell B_\ell(x)\big)_x$, with adjoint
$(M^\top H)_\ell=\sum_x A_\ell(x)^\top H_x B_\ell(x)^\top$; target $\Dstar:=(-\eta G(x))_x$;
$\Pi$ the projector onto $\mathrm{range}(M)$. By N3, $\dW^{\mathrm{GD}}\propto M^\top\Dstar$.

\paragraph{N5. Transfer operator and gain.}
$T(H):=\sum_\ell A_\ell A_\ell^\top H B_\ell^\top B_\ell$, so that the induced operator change is
$-\eta\,T(G)+O(\eta^2)$. With the SVD $P=\sum_k s_k u_kv_k^\top$, the \textbf{mode gain} is
\begin{equation}\label{eq:gain}
  c_k := \big\langle u_k,\ T(u_kv_k^\top)\,v_k\big\rangle
       = \sum_{\ell}\big\|A_\ell^\top u_k\big\|^2\big\|B_\ell v_k\big\|^2 .
\end{equation}

\paragraph{N6. $\eta$ is an operator step, not a learning rate.} $\Dstar=-\eta G$ asks the
operator to move by $\eta\|G\|$, so $\eta$ reads directly as a fraction of $\|P(x)\|$ and is
comparable across depths and architectures --- unlike a learning rate. \NOTE{At $d{=}32$,
$L{=}8$: $\eta=0.1$ requests a $14\%$ change of the operator per step, $\eta=1$ a $143\%$ one.}

\begin{definition}[what the diagnostic reports]\label{sec:diag}
\begin{enumerate}[label=(D\arabic*),leftmargin=*,itemsep=0pt]
  \item \textbf{Reachability} $\alpha=\|\Pi\Dstar\|/\|\Dstar\|$ --- how much of the ideal the
        architecture admits. Calibration, not a finding.
  \item \textbf{Efficiency} $\epsilon=\cos(MM^\top\Dstar,\ \Pi\Dstar)$ --- how much of the
        \emph{reachable} step gradient descent takes.
  \item \textbf{Gain exponent} $d\log c_k/d\log s_k$ and its intercept, from \eqref{eq:gain}.
\end{enumerate}
\end{definition}

\NOTE{\textbf{The probe matters.} $c_k$ must be measured with the rank-one direction
$u_kv_k^\top$. Probing with a random $H$ and taking $\langle u_k,T(H)v_k\rangle/\langle
u_k,Hv_k\rangle$ equals $c_k$ only when $A_\ell A_\ell^\top$ and $B_\ell^\top B_\ell$ are
diagonalised by the singular vectors of $P$ --- true under balancedness, \emph{false} otherwise.
An earlier version of this work used the random probe and reported a qualitatively wrong
exponent; see \S\ref{sec:withdrawn}.}

\section{The questions, and what is already known about each}\label{sec:questions}

\NOTE{Checked against the literature 2026-09-18. Each question states what is settled and what is
not, so that a result here cannot restate a known one.}

\paragraph{Q1 --- Does a per-input gain law hold?}
In DLN the entire bias reduces to one exponent: $c_k=L\,s_k^{2-2/L}$ \cite{arora2018optimization}.
\emph{Question:} for each input $x$, does the realised gain
$c_k(x)=\big(u_k^\top\Delta \Jop(x)v_k\big)/\big(u_k^\top G(x)v_k\big)$ obey a power law in
$s_k(x)$, and with what exponent?
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \textbf{Known.} The \emph{rank} question is settled in the negative: GF on ReLU need not
        minimise rank even approximately \cite{timor2023implicit}, though sufficient depth does
        induce low-rank bias in some settings. Infinite depth with regularisation gives the
        bottleneck rank \cite{jacot2023implicit}; the first escape from the origin is low-rank in
        deeper layers \cite{bantzis2025saddle}; effective rank falls with depth empirically
        \cite{huh2021lowrank}. For \emph{fixed} gates, \cite{haas2026why} obtains $2-2/L$ under
        assumptions it labels deliberately strong.
  \item \textbf{Not known.} All of the above concern \emph{endpoints, limits, or ranks}. The
        finite-time \emph{rate} --- the exponent governing per-input singular-value motion with
        \emph{input-dependent} gates --- is not addressed by any of them. \textbf{Q1 must be posed
        as a question about the exponent, not about rank}, or it restates
        \cite{timor2023implicit}.
\end{itemize}

\paragraph{Q2 --- Is the bias shared across inputs, or per-input?}
\emph{Question:} do the dominant subspaces of $\Jop(x)$ align across $x$ during training?
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \textbf{Known, and this largely settles it.} The Neural Feature Ansatz
        \cite{radhakrishnan2024mechanism,beaglehole2024agop} states that each layer's weight Gram
        matrix tracks the average gradient outer product --- an average \emph{over inputs}, hence a
        shared-structure claim, across architectures and tasks.
        \cite{wang2016extended} already studies the spectrum of the stacked per-input Jacobian.
  \item \textbf{Not known.} Whether the \emph{per-input operator} spectra (as opposed to the
        weight Grams) align, and whether that alignment is the \emph{mechanism} producing the Q1
        exponent rather than a correlate. \textbf{Demote Q2 to a check on Q1's framing; it is not
        a contribution on its own.}
\end{itemize}

\paragraph{Q3 --- Does efficiency explain the deviation?}
\emph{Question:} if the Q1 exponent departs from $2-2/L$, is the departure predicted by
$\epsilon$ (\S\ref{sec:diag})?
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \textbf{Known.} Nothing found. The closest is \cite{pennington2017resurrecting}: ReLU
        cannot achieve dynamical isometry while other activations can --- which is the same
        phenomenon as our $\alpha$ gap, at initialisation and for $\Jop$ rather than for the step
        map $M$. \textbf{Our $\alpha=1$ for CReLU versus $\alpha<1$ for ReLU must be presented as
        the operator-step counterpart of that known fact, not as a discovery.}
  \item \textbf{Not known, and this is the open one.} A quantitative bridge from the linear law to
        the nonlinear regime through a measurable quantity. It is conditional on Q1 producing a
        deviation to explain.
\end{itemize}

\paragraph{Consequence for the contribution.} \TD{} C5 should be Q1 (as a rate question) and Q3.
Q2 is a validity check. Any claim about rank, or about ReLU's poor conditioning, is already in the
literature above.

\section{The decomposition}\label{sec:decomp}

\NOTE{Bookkeeping, not a theorem: a definition, Cauchy--Schwarz, a substitution, and the linearity
of least squares. Its value is that it says \emph{which} quantity carries the deviation from
$2-2/L$, so a measured shortfall can be attributed rather than merely reported.}

Fix a direction $k$ and write $a_\ell:=A_\ell^\top u_k$, $b_\ell:=B_\ell v_k$, so that
\eqref{eq:gain} reads $c_k=\sum_\ell\|a_\ell\|^2\|b_\ell\|^2$. Because
$P=A_\ell W_\ell B_\ell$ holds for \emph{every} $\ell$,
\begin{equation}\label{eq:cs}
  s_k \;=\; u_k^\top P v_k \;=\; a_\ell^\top W_\ell\, b_\ell
      \;\le\; \|a_\ell\|\,\|W_\ell\|_2\,\|b_\ell\| ,
\end{equation}
so the ratio
\begin{equation}\label{eq:r}
  r_{\ell,k} \;:=\; \frac{\|a_\ell\|\,\|b_\ell\|\,\|W_\ell\|_2}{s_k}\;\ge\;1
\end{equation}
measures the slack in \eqref{eq:cs}: it is $1$ exactly when $(a_\ell,b_\ell)$ is the top singular
pair of $W_\ell$. Substituting $\|a_\ell\|^2\|b_\ell\|^2=(r_{\ell,k}s_k/\|W_\ell\|_2)^2$ into
\eqref{eq:gain} gives, with $R_k:=\sum_\ell\big(r_{\ell,k}/\|W_\ell\|_2\big)^2$,
\begin{equation}\label{eq:decomp}
  \boxed{\;c_k \;=\; s_k^{2}\,R_k\;}
  \qquad\Longrightarrow\qquad
  \log c_k \;=\; 2\log s_k + \log R_k \quad\text{pointwise in }k .
\end{equation}
Least squares is linear in the response, so fitting $\log c$ on $\log s$ across directions yields
\begin{equation}\label{eq:slope}
  \underbrace{\frac{d\log c}{d\log s}}_{\text{gain exponent}}
  \;=\; 2 \;+\; \frac{d\log R}{d\log s}.
\end{equation}

\paragraph{The balanced case.} Under balancedness \cite{arora2018optimization}
$A_\ell A_\ell^\top=(PP^\top)^{\frac{L-\ell}{L}}$ and
$B_\ell^\top B_\ell=(P^\top P)^{\frac{\ell-1}{L}}$, so
$\|a_\ell\|^2=s_k^{2(L-\ell)/L}$, $\|b_\ell\|^2=s_k^{2(\ell-1)/L}$ and
\begin{equation}\label{eq:bal}
  c_k = L\,s_k^{2-2/L},\qquad R_k = L\,s_k^{-2/L},\qquad
  \frac{d\log R}{d\log s} = -\frac{2}{L},\qquad
  \min_\ell r_{\ell,k} = \Big(\tfrac{s_{\max}}{s_k}\Big)^{1/L}.
\end{equation}
So $r>1$ even when balanced: the slack is structural, growing as the singular value shrinks, and
$-2/L$ is precisely the rate that produces $2-2/L$ through \eqref{eq:slope}.

\paragraph{Verified.} \RUN{} $d{=}24$, $L\in\{4,16\}$: \eqref{eq:decomp} holds to
$1$--$3\times10^{-16}$ \emph{at any initialisation} (it is algebra); the balanced predictions
\eqref{eq:bal} hold to $3\times10^{-15}$ when balanced and fail by $0.6$--$16$ at Xavier, as they
must.

\section{The solvers}\label{sec:solver}

\paragraph{S1 --- exact minimal realisation (one target operator).}
$\min_{\dW}\sum_\ell\|\dW_\ell\|_F^2$ s.t. $\prod_\ell(W_\ell+\dW_\ell)=P_\star$. Split the chain
at its midpoint, solve the two-factor problem from the closed-form SVD split $Y_0=US^t$,
$X_0=S^{1-t}V^\top$ with $t=a/(a+b)$, and recurse. \OK{} Exact from a degenerate base point for
every $L$ with the depth-proportional exponent (Thm~\ref{thm:exact}); verified to $1.000000$ for
$L=2..128$ at $d{=}32$ in closed form. \TD{} Away from degeneracy it is a heuristic
($\rho=0.05$ at $L{=}8$ to $0.23$ at $L{=}32$).

\paragraph{S2 --- truncated Krylov on the step map (input-dependent gates).}
With per-input operators there is no single target, so we solve
$\min_{\dW}\|M\dW-\Dstar\|^2$ by LSQR truncated at $k$ iterations, over the Krylov space
$\mathcal K_k=\mathrm{span}\{M^\top\Dstar,(M^\top M)M^\top\Dstar,\dots\}$. At $k=1$ this is the
gradient direction with the residual-minimising step length (measured:
$\cos(\dW^{(1)},-\nabla)=+1.0000$); as $k$ grows it approaches the reachable ideal $\Pi\Dstar$.
\NO{} \textbf{Linearised, hence stationary at $W=0$} --- the case S1 exists for.
\TD{} No exact-constraint analogue of S1 for a family of targets: the recursion splits \emph{one}
target and per-input intermediate targets conflict.

\paragraph{Cost.} \RUN{} GPU implementation (\texttt{studies/gpu.py}, float32; diagnostics stay
CPU/float64) gives $5\times$ at $L{=}8$, $d{=}32$ up to $41\times$ at $L{=}32$, $d{=}64$,
batch $256$, $k{=}100$ --- $32.5$\,s to $0.8$\,s per step. The gain grows with size because the
GPU is launch-bound at these shapes.

\section{The comparison protocol}\label{sec:protocol}

Having a reference step makes a comparison possible that endpoint studies cannot make. The two
modes answer different questions and both are needed.

\paragraph{Mode A --- one trajectory, every step scored.}
A single optimiser drives the trajectory. At probe steps we stop and, \emph{from those weights},
ask what each candidate would do. With $M$, $\Dstar$ and $\Pi$ as in N4, write
$\dW_{\mathrm{id}}:=M^{+}\Dstar$ for the minimum-norm weight update realising the reachable ideal
$\Pi\Dstar=M\dW_{\mathrm{id}}$. For a candidate step $\dW_m$ we record
\begin{align*}
  \cos_{\mathrm{op}} &= \cos\!\big(M(\dW_m),\ \Pi\Dstar\big),
  &\cos_{\mathrm{w}} &= \cos\!\big(\dW_m,\ \dW_{\mathrm{id}}\big),\\
  \mathrm{ratio}_{\mathrm{op}} &= \|M(\dW_m)\|/\|\Pi\Dstar\|,
  &\mathrm{ratio}_{\mathrm{w}} &= \|\dW_m\|/\|\dW_{\mathrm{id}}\|.
\end{align*}
Nothing differs between candidates except the update rule --- same weights, same batch, same
target. \textbf{The two cosines are scale-invariant, so they do not depend on the learning rate at
all}; only the ratios do, which is why they are reported separately. Adam is stateful, so its
moments are carried along the reference trajectory at every step and ``Adam'' here means what Adam
would do having followed this path --- exactly the real thing when Adam is the reference.

\paragraph{Fidelity is not quality --- and must be said before the numbers.}
$\Dstar=-\eta\,\partial\mathcal L/\partial P$ is \emph{steepest descent in operator space}. On an
anisotropic loss steepest descent is an indirect route: it removes the steep direction first and
then crawls along the valley. So a method that scores \emph{lower} on $\cos_{\mathrm{op}}$ can
reach the optimum \emph{sooner} --- and Adam does, in E8, because per-coordinate normalisation
approximates a whitened step. $\cos_{\mathrm{op}}$ measures how far the factorisation deflects the
step the loss asked for; it is not an optimiser ranking, and must not be read as one.
\NOTE{Measured, so the claim is not rhetorical: in E8 at $L{=}32$ Adam reaches
$\|P-P^\star\|=0.0000$ against the reference's $0.1437$ while scoring a mean alignment of $+0.44$
against the reference's $+0.99$.}

\paragraph{The deflection is signed, and Adam's changes sign.} Measuring
$\cos(\Delta P,\,-\partial\mathcal L/\partial P)$ at every step of E8, Adam moves the operator
\emph{against} the direction the loss asked on $6\%$ of steps at $L{=}2$ and $19$--$28\%$ at
$L{=}32$, reaching $-0.999$; gradient descent and the reference never do. Depth makes it worse.
\NOTE{The reference is faithful to $+0.999$ once $\eta\le0.02$; at $\eta=0.06$ it reads $0.975$
purely because the step leaves the linear regime, not because it is less faithful. Mode A avoids
this confound entirely by scoring the first-order $M(\dW)$ rather than the realised $\Delta P$.}

\paragraph{Mode B --- free-running.} Each optimiser takes its own course; endpoints are compared at
matched training loss on loss, accuracy, and the effective rank of $P(x)$. This is what E4 does.
Mode B alone cannot attribute a difference to the update rule, because by any late step the arms
sit at different weights; Mode A supplies that attribution.

\NOTE{\textbf{Implementation note that cost real time.} $\dW_{\mathrm{id}}$ needs a converged
least-squares solve, and LSQR on these step maps \emph{diverges once it has converged} --- on a
depth-4 ReLU net the iterate reaches $10^{16}$ by $k{=}2000$ having converged by $k{\approx}200$.
Two natural stopping tests both fail: LSQR's internal \texttt{phibar} is monotonically
non-increasing by construction and never signals anything, and $\|M\dW-\Dstar\|$ is blind to the
iterate drifting into $\ker M$ (measured: $\|\dW\|=2.9\cdot10^9$ with the residual unchanged to
seven digits). The normal-equation residual $\|M^\top(M\dW-\Dstar)\|$ detects both and is the
stopping rule used; it is non-monotone, so the best iterate is retained rather than the last. Its
achieved value is recorded at every probe, because every number in Mode A is measured against
$\dW_{\mathrm{id}}$ and is meaningless if that solve is bad.}

\NOTE{\textbf{Depth limitation, measured.} $\cos_{\mathrm{op}}$ is reliable at every depth tested;
$\cos_{\mathrm{w}}$ is not. At $L{=}16$ two solves realising the same operator change to
$\cos 0.976$--$0.997$ can have near-orthogonal weight vectors (deep linear, $L{=}16$:
$\cos_{\mathrm{w}}=0.045$) --- they are both valid, differing by an element of $\ker M$, so the
comparison is between arbitrary choices inside the solution set rather than between methods. This
is \emph{not} caught by the solve-quality flag: that cell has a normal-equation residual of
$2.8\cdot10^{-9}$. At $L\in\{4,8\}$ all four architectures give $\cos_{\mathrm{w}}=1.000000$.
Accordingly the E7 grid reports $\cos_{\mathrm{op}}$ as the directional result and restricts
$\cos_{\mathrm{w}}$ to shallow depth.}

\section{Supporting results}\label{sec:thms}

\begin{theorem}[degeneracy]\label{thm:degen}\OK{}\RUN{}
At $W=0$ the product map is homogeneous of degree $L$, so all derivatives of order $<L$ vanish and
every null-consistent derivative-based rule of order $p<L$ returns $\dW=0$ --- gradient descent,
Adam, Muon, Shampoo, K-FAC, Newton, Gauss--Newton, natural gradient with pseudo-inverse Fisher
\cite{bernacchia2018exact}, and a first-order solver applied to the exact objective.
\end{theorem}

\begin{theorem}[exact recursion]\label{thm:exact}\OK{}\RUN{}
From a degenerate base point S1 solves the realisation problem exactly for every $L$ iff each
split uses $t=a/(a+b)$; the symmetric split $t=\tfrac12$ is exact iff $L$ is a power of two.
\end{theorem}

\begin{theorem}[conservation]\label{thm:conserve}\OK{}\RUN{}
Any flow $\dot W_\ell=A_\ell^\top\Lam B_\ell^\top$ with a \emph{shared} $\Lam$ conserves
$W_{\ell+1}^\top W_{\ell+1}-W_\ell W_\ell^\top$. Balancedness
\cite{du2018algorithmic,arora2018optimization} is therefore a property of minimal realisations,
not of gradient descent. \NOTE{Measured: $1.5\times10^{-15}$ for random shared $\Lam$ against
$1.4\times10^{1}$ for an independent per-layer step.}
\end{theorem}

\NOTE{Also established: FGLN gradient flow is deep-linear flow in the variables
$M_\ell=D_\ell W_\ell D_{\ell-1}$, so \cite{arora2018optimization} Thm~1 transfers to FGLN under
$M$-balancedness and \cite{haas2026why} Prop.~4.5 is correct.}

\section{Experiments}\label{sec:exp}

\subsection*{Part I --- deep linear: the framework and its validation}

\paragraph{E1 --- Validation. \OK{}\RUN{}}
With a \emph{balanced} chain carrying a prescribed non-flat spectrum, the measured exponent equals
$2-2/L$ and the intercept equals $L$, both to $10^{-14}$, for $L=2..32$ --- and the same holds for
a \emph{balanced CReLU} network (looks-linear blocks built from that spectrum, so the gate absorbs
and $P$ stays non-flat). The pipeline therefore reproduces \cite{arora2018optimization} Thm~1 in
the linear \emph{and} the nonlinear implementation.
\NOTE{Two traps worth stating in the paper. An \emph{orthogonal} chain is balanced but has a
\emph{flat} spectrum, so no exponent is identifiable from it: balanced $\neq$ orthogonal. And with
more than one input the per-input ratio is not $Ls^{2-2/L}$, because one weight step answers the
aggregate target.}

\paragraph{E2 --- The exponent at standard initialisation. \RUN{}}
He initialisation, width 32, 2 seeds, 8 inputs, rank-one probe.

deep linear  & 0.733 & 1.093 & 1.318 & 1.584 & 1.653 \\

The exponent \emph{rises with depth} as $2-2/L$ predicts but sits systematically below it by

By \eqref{eq:slope} the shortfall is exactly the excess steepness of $d\log R/d\log s$

\subsection*{Part II --- nonlinear: what changes when the ideal is not reachable}

\paragraph{E3 --- Reachability and efficiency. \RUN{}}
$L{=}8$, width 64, MNIST-1D, 2 seeds, 300 Adam steps. CReLU holds $\alpha=1.0000\pm0.0000$ at
every step with LSQR converging to $10^{-9}$; ReLU falls from $0.980$ to $0.924$ and its LSQR does
\emph{not} converge at 3000 iterations (residual $0.19\to0.38$), so its $\alpha$ is a lower bound
and its $\epsilon$ unreliable.
\NOTE{$\alpha$ is the operator-step counterpart of the known fact that ReLU cannot achieve
dynamical isometry \cite{pennington2017resurrecting}: calibration, not a finding. That ReLU's step
map resists Krylov by three orders of magnitude more than CReLU's is the substantive contrast.}

\paragraph{E4 --- Endpoints. \RUN{} seed 2 running}
Adam at the reference MNIST-1D recipe ($L{=}8$, width 32, 4000/1000 split, batch 100,
lr$\,\in\{3\cdot10^{-3},10^{-2}\}$) against the reachable ideal ($k{=}50$,
$\eta\in\{0.1,0.3,0.5\}$), compared \emph{at matched training loss} from the recorded
trajectories. Effective rank is the participation ratio $(\sum_i\sigma_i)^2/\sum_i\sigma_i^2$ of
$P(x)$, averaged over inputs; accuracy is at the same matched loss.

\begin{center}\small
\begin{tabular}{llrrrr}
\toprule
 & & \multicolumn{2}{c}{Adam} & \multicolumn{2}{c}{reachable ideal} \\
\cmidrule(lr){3-4}\cmidrule(lr){5-6}
arch & seed & rank & acc & rank & acc \\
\midrule
CReLU & 0 & 5.19 & 0.550 & 7.23 & 0.540 \\
CReLU & 1 & 5.15 & 0.533 & 6.93 & 0.551 \\
ReLU  & 0 & 3.24 & 0.455 & 5.48 & 0.465 \\
\bottomrule

\end{center}

\textbf{At matched training loss the two arms reach the same accuracy but systematically different
effective rank}, the operator arm higher by $1.7$--$2.2$, replicated across seeds and both
architectures. Everything but the update rule is held fixed, so this is a controlled counterfactual
rather than the usual correlational read across depth, width or initialisation.
\NOTE{Direction confirmatory --- that gradient descent is low-rank biased is established
\cite{arora2019implicit,gunasekar2017implicit}. What is new here is only the control. Also: no
dose--response in $\eta$; the ReLU ordering reverses ($5.69/4.91/3.90$ for
$\eta{=}0.1/0.3/0.5$), so $\eta$ is not a monotone dial on the bias and must not be sold as one.}

\NOTE{\TD{}\textbf{LR fairness --- not yet satisfied, and this is the table's weak point.} The grid
was lr$\,\in\{3\cdot10^{-3},10^{-2}\}$, and $10^{-2}$ plateaus above the matched-loss threshold
(train $0.345$), so \emph{every Adam entry above is $3\cdot10^{-3}$}: the baseline is effectively
untuned, at one learning rate, and neither $10^{-3}$ nor $10^{-4}$ --- the conventional choices ---
was run. This is the same objection that the previous submission was rejected on (a baseline at
$10^{-3}$ against the method at lr $1.0$), so it must be closed before the rank gap is claimed.
Extension to lr$\,\in\{10^{-4},3\cdot10^{-4},10^{-3}\}$ at all three seeds is queued; Adam cells
cost ${\sim}15$\,s each, so there is no excuse for the omission.}

\paragraph{E6 --- Does the bias ever pay? \NO{} withdrawn}
The design was sound and the answer is negative. Teacher $y=W_2\,\mathrm{relu}(W_1x)$ with
$W_1\in\R^{r\times32}$, so the target Jacobian has rank at most $r$; student a ReLU MLP at $L{=}8$;
arms compared at matched training loss. A first pass showed the operator arm generalising
$3.7\times$ better at $r{=}1$, decaying to parity by $r{=}8$. Two controls removed it.

\emph{The Adam grid was truncated above.} Adam's best learning rate was the largest one tried at
every $r$, which is the signature of a grid that has not bracketed its optimum. Extending it to
$\{3\cdot10^{-2},10^{-1}\}$ erases the effect where it was largest: at $r{=}1$ the ratio falls from
$3.72$ to $0.82$ and at $r{=}2$ from $3.07$ to $1.04$ --- Adam at lr $0.1$ is slightly \emph{ahead}.

\emph{And the rest is seed noise.} Across seeds $0/1/2$ the ratio at $r{=}1$ is $3.72/0.82/0.12$
and at $r{=}4$ is $1.64/1.33/0.84$: both flip sign, the first across a factor of $30$. Only
$r{=}2$ replicated ($3.07/2.65/2.26$), and that is precisely the point the extended grid drops to
parity.

\NOTE{Two things are worth keeping from the wreckage. First, the hypothesis that a \emph{smaller}
Adam learning rate would stabilise and close the gap is refuted, not merely untested: at $10\times$
the budget (40k steps) lr $\in\{10^{-4},3\cdot10^{-4},10^{-3}\}$ give $0.341/0.262/0.187$ at
$r{=}1$ against $0.052$ for lr $10^{-2}$ at 4k steps --- monotonically worse, not better. Second,
an optimisation gap does survive and is large: at $r{=}16$ and $r{=}32$ Adam stalls at train
$0.242$ and $1.520$ within 4000 steps while the operator arm reaches $0.018$ and $0.586$
($13.6\times$ and $2.6\times$ lower). That is a statement about fitting, not generalisation, and
belongs in Part II as such.}

\NOTE{\textbf{Procedural lesson, applies to E4 as well.} Both failures here --- a best-over-grid
statistic read before the grid was complete, and a best-over-grid statistic read from a grid that
never bracketed the baseline's optimum --- are failures of the \emph{protocol}, not of the
measurement. E4 currently rests on an Adam grid whose only surviving point is lr $3\cdot10^{-3}$;
until the extended grid returns, its rank gap must be treated as provisional for exactly the reason
E6 fell.}

\paragraph{E7 --- The Mode A grid. \TD{} prober written and validated, not yet run}
The measurement the new framing is built around (\S\ref{sec:protocol}), over
models $\times$ initialisations $\times$ optimisers: deep linear, FGLN, ReLU and CReLU;
Xavier (random, ill-conditioned at depth) and orthogonal (looks-linear for CReLU, which is the
reason that architecture is in the paper at all --- it gives a \emph{nonlinear} network that is
perfectly conditioned at any depth, so conditioning and gating can be varied independently);
gradient descent, Adam, and the reference at several truncation levels $k$. Endpoints additionally
report train/test loss, train/test accuracy and the effective rank of $P(x)$.
\NOTE{Validated in \texttt{studies/probe\_steps.py}: the $k{=}1$ reference step is parallel to the
gradient step to $10^{-6}$ for all four architectures (it must be --- both are $M^\top\Dstar$),
large $k$ reproduces $\dW_{\mathrm{id}}$ and $\Pi\Dstar$, and $\Dstar-\Pi\Dstar\perp\Pi\Dstar$
confirms $\Pi$ is the projector. \TD{} FGLN still needs a fixed-gate path in
\texttt{studies/gpu.py}; the other three are in.}

\paragraph{E8 --- Operator-space trajectories. \RUN{}}
A deep linear chain $W_L\cdots W_1=P\in\R^{1\times2}$ on a two-feature regression, so the operator
has exactly two components and its trajectory is \emph{drawn}, not projected, while the weight
space stays large ($1944$ parameters at $L{=}32$, $h{=}8$). The loss is quadratic in $P$, so the
contours are exact and $P^\star$ is the OLS solution. Gradient descent, Adam, ALS and the minimal
reference, at $L{=}2$ and $L{=}32$. Every arm descends the \emph{same} quadratic in $P$; all that
differs is how the factorisation turns a weight step into an operator step, so every separation
visible is attributable to the parameterisation rather than to the objective.
\NOTE{\textbf{The plane matters, and the obvious choice is wrong.} A model whose \emph{weight}
space is two-dimensional (a scalar two-layer chain) has a rank-one step map $M=[w_2,w_1]$, so
$M^{+}\propto M^\top$ and gradient descent and the minimal reference are collinear --- measured at
$\cos=1.0000000000$. Such a figure cannot show a direction difference at all. The previous
submission's figure separated because it plotted the \emph{operator}, where $M$ has rank two.
Separately: ALS needs ridge damping here; an undamped per-layer exact sweep is anti-aligned with
the operator gradient on $100\%$ of steps at $L{=}2$ and diverges to $\|P-P^\star\|=154$.}

\section{Withdrawn, and not claimed}\label{sec:withdrawn}

\begin{itemize}[leftmargin=*,itemsep=2pt]
  \item \NO{} \textbf{``The exponent saturates near $1$ rather than following $2-2/L$.''}
        An artefact of probing the gain with a random $H$ instead of $u_kv_k^\top$; that ratio
        equals $c_k$ only under balancedness. Corrected in E2.
  \item \NO{} \textbf{``The nonlinearity suppresses the gain magnitude by more than an order of
        magnitude.''} An artefact of using He initialisation ($\sqrt{2/d}$) for the \emph{deep
        linear} baseline, where the variance-preserving scale is $\sqrt{1/d}$; He makes the linear
        product grow like $2^{L/2}$. With the correct baseline the intercepts at $L{=}32$ are
        $357$ (linear), $467$ (CReLU), $560$ (ReLU) --- the nonlinear networks are \emph{higher}.
        The exponent is unchanged by the scale ($1.8000$ either way), as it must be.
  \item \NO{} \textbf{``Gates break the conservation law.''} Absorbing the gates into
        $M_\ell=D_\ell W_\ell D_{\ell-1}$ makes the FGLN flow literally the linear flow;
        \cite{haas2026why} Prop.~4.5 is correct.
  \item \NO{} $K_\ell\ge L$ is not hypothesis-free: Lemma B gives only
        $K_\ell\ge L(s_\ell/g)^{2/L}$, and the AM--GM step is in \cite{rawal2026saddle}.
  \item \textbf{Not claimed}: gate-induced rank deficiency (self-evident, used to calibrate
        $\alpha$); any new characterisation of the deep linear bias
        (\cite{arora2018optimization}); a benchmark win --- S2 realises at best ${\sim}80\%$ of its
        target and costs $4\times$ Adam.
  \item \textbf{The decomposition of \S\ref{sec:decomp} is bookkeeping}, not a theorem: a
        definition, Cauchy--Schwarz, a substitution and the linearity of least squares, with the
        balanced case a restatement of \cite{arora2018optimization}. Its content is that it
        localises the deviation, and the empirical fact that $d\log R/d\log s$ is steeper than
        $-2/L$ in practice.
\end{itemize}

\section{Open items}\label{sec:open}
\begin{enumerate}[leftmargin=*,itemsep=0pt]
  \item Re-measure the exponent \emph{along training} with the rank-one probe (voids the current
        training-time numbers). Blocks the first question of the abstract.
  \item \textbf{Tune the Adam baseline in E4 before claiming the rank gap.} Every Adam entry is
        currently one learning rate, $3\cdot10^{-3}$; $\{10^{-4},3\cdot10^{-4},10^{-3}\}$ are
        queued. If the rank gap shrinks at the conventional learning rates, E4 is not a result.
        Same discipline everywhere: report the grid, and say which point in it each number came
        from.
  \item Finish E4 (seed 2). The rank gap is replicated and the accuracy null is stable; a null
        needs more evidence than a positive, so the third seed decides how it is stated.
  \item Finish E6 ($r\in\{16,32\}$, then seeds 1--2), and settle the two grid objections: extend
        Adam to $\{10^{-4},3\cdot10^{-4}\}$ at a $40$k-step horizon (the small-lr hypothesis) and
        to $\{3\cdot10^{-2},10^{-1}\}$ (the grid is truncated above). Until then the crossover is
        an observation, not a result.
  \item \NO{} E6 is withdrawn (truncated grid + seed noise); the leaky-ReLU mechanism run behind it
        is now moot as a mechanism study, though its ReLU arm is still a useful control. The
        surviving question from that line is the \emph{optimisation} gap at high target rank.
  \item [superseded] Explain E6's mechanism or drop it. The gap is not rank and not early stopping; gate death
        is the current suspect and is only a correlation. The real-data counterpart (MNIST-1D
        denoising onto a rank-$r$ PCA target, \texttt{studies/rank\_sweep.py --task mnist1d}) is
        written but unlaunched, and may be near-linear enough that both arms agree --- which would
        be the mechanism predicting its own boundary, not a null.
  \item Re-verify the $k\to\infty$ exponent with the rank-one probe (second question).
  \item Port E5 to the GPU path and extend to $L=64$ with seeds.
  \item Close the S1/S2 gap, or state the restriction.
\end{enumerate}
