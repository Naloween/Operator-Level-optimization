\section{Question, and why this framing}

\textbf{Question.} What is the implicit bias of gradient descent on a factored model, measured
\emph{at the level of the operator the model computes}, and how does the answer change when the
model becomes nonlinear?

\textbf{Framing.} A depth-$L$ network computes a linear operator per input,
$f(x)=\Jop(x)x$. Gradient descent updates \emph{weights}; the operator changes as a consequence.
We take the \textbf{operator gradient step} $-\eta G(x)$, $G=\partial\ell/\partial \Jop$, as the
reference against which the realised change $\Delta \Jop(x)$ is measured. The bias is the
discrepancy. This framing is useful because the reference is architecture-independent: the same
target is defined for a linear network, a fixed-gate network, and a ReLU network, so the
\emph{same} diagnostic runs across all three and its readings are comparable.

\textbf{Programme.} (i) Validate in the deep linear case, where the answer is known in closed
form, and require agreement. (ii) Extend to fixed gates, where the closed form still exists.
(iii) Extend to input-dependent gates (ReLU, CReLU), where it does not, and report what the
diagnostic finds. The contribution is what (iii) shows, not the machinery.

\paragraph{Contributions.} \NOTE{Methods count as contributions where they are new, but their
weight depends on what they show; C5 is what the rest is for.}
\begin{enumerate}[label=\textbf{C\arabic*.},leftmargin=*,itemsep=1pt]
  \item \textbf{An operator-level diagnostic with the correct normalisation}, valid unchanged
        across DLN, FGLN, ReLU and CReLU: reachability $\alpha$, efficiency $\epsilon$, spectral
        effect (Def.~\ref{def:diag}). Separating $\alpha$ from $\epsilon$ is what makes readings
        comparable across architectures; the unnormalised version conflates them.
        \OK{} \RUN{}
  \item \textbf{Gradient descent is exactly one Landweber step} of the operator least-squares
        problem, and truncated Krylov iteration is a calibrated dial between it and the best
        achievable step (Prop.~\ref{prop:landweber}, \S\ref{sec:solver} S2). Verified:
        $\cos(\dW^{(1)},-\nabla\mathcal L)=+1.0000$. \OK{} \RUN{}
  \item \textbf{Balancedness is a property of minimal realisations, not of gradient descent}
        (Thm~\ref{thm:conserve}): any step of the shared-multiplier form conserves the Du--Hu--Lee
        invariant. \OK{} \RUN{}
  \item \textbf{A constructive solver for the exact realisation problem} that works where every
        derivative-based method provably cannot move, with an exactness characterisation: the
        hierarchical recursion is exact for every $L$ iff each split uses the depth-proportional
        exponent (Thms~\ref{thm:degen}, \ref{thm:exact}). \OK{} \RUN{}
  \item \textbf{What the diagnostic finds in the nonlinear regime.} \TD{} The gap named in
        \S\ref{sec:sota}. This is the contribution the others exist to support;
        \S\ref{sec:exp} E2--E3.
\end{enumerate}

\NOTE{Not claimed: gate-induced rank deficiency (self-evident, used only to calibrate $\alpha$);
any new characterisation of the DLN bias; a benchmark win.}

\section{Notation}\label{sec:notation}

\NOTE{Every symbol used later is defined here, in order of dependence.}

\paragraph{N1. Model classes.} Weights $W_\ell\in\R^{n_\ell\times n_{\ell-1}}$, $\ell=1,\dots,L$,
written collectively $W=(W_1,\dots,W_L)$. Input $x\in\R^{n_0}$, output in $\R^{n_L}$.
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \textbf{DLN} (deep linear): $\Jop = W_L\cdots W_1$, independent of $x$.
  \item \textbf{FGLN} (fixed gates): $\Jop = W_LD_{L-1}\cdots D_1W_1$ with $D_\ell$ a fixed
        diagonal matrix, independent of $x$.
  \item \textbf{ReLU MLP} (no biases): pre-activations $z_\ell(x)$, gates
        $D_\ell(x)=\mathrm{diag}(\mathbb 1[z_\ell(x)>0])$, and
        $\Jop(x)=W_LD_{L-1}(x)\cdots D_1(x)W_1$, so that $f(x)=\Jop(x)\,x$ exactly.
  \item \textbf{CReLU MLP}: features $c=[\mathrm{relu}(z);\mathrm{relu}(-z)]$,
        $W_\ell\in\R^{d\times 2d}$, gate
        $D(z)=\big[\mathrm{diag}(\mathbb 1[z>0]);-\mathrm{diag}(\mathbb 1[z<0])\big]\in\R^{2d\times d}$.
        Since $D(z)^\top D(z)=I$ the gate is an isometry, and the looks-linear initialisation
        $W=[O\mid -O]$ gives $W D(z)=O$ for every $z$, hence $\Jop(x)=O_L\cdots O_1$ orthogonal at
        any depth.
\end{itemize}

\paragraph{N2. Contexts.} For each $\ell$ define $A_\ell(x)$ (everything to the left of $W_\ell$)
and $B_\ell(x)$ (everything to the right) by
\[
  B_1=I,\quad B_{\ell+1}=D_\ell(x)W_\ell B_\ell(x),\qquad
  A_L=I,\quad A_\ell=A_{\ell+1}(x)W_{\ell+1}D_\ell(x),
\]
so that $\Jop(x)=A_\ell(x)\,W_\ell\,B_\ell(x)$ for every $\ell$. (DLN and FGLN are the cases where
$A_\ell,B_\ell$ do not depend on $x$.)

\paragraph{N3. Loss and operator gradient.} Per-sample loss $\ell(f(x),y)$, total
$\mathcal L=\frac1n\sum_x \ell$. The \emph{operator gradient} is
$G(x):=\partial \ell/\partial \Jop(x)\in\R^{n_L\times n_0}$; it is rank one both for the squared
loss ($G=(f-y)x^\top$) and for cross-entropy ($G=(\mathrm{softmax}(f)-e_y)x^\top$). The
coordinate gradient follows by the chain rule:
\begin{equation}\label{eq:coordgrad}
  \frac{\partial\mathcal L}{\partial W_\ell}=\sum_x A_\ell(x)^\top G(x) B_\ell(x)^\top .
\end{equation}

\paragraph{N4. The two spaces and the step map.} Let
\[
 \mathcal W=\bigoplus_{\ell=1}^L\R^{n_\ell\times n_{\ell-1}}\ \ (\text{weight perturbations}),
 \qquad
 \mathcal O=\bigoplus_{x\in\mathcal B}\R^{n_L\times n_0}\ \ (\text{per-input operator changes}),
\]
both with the Frobenius inner product, $\mathcal B$ the batch, $|\mathcal B|=n$. Define the
linear \textbf{step map} $M:\mathcal W\to\mathcal O$ and its adjoint by
\begin{equation}\label{eq:Mdef}
  \big(M\dW\big)_x:=\sum_{\ell=1}^L A_\ell(x)\,\dW_\ell\,B_\ell(x),
  \qquad
  \big(M^\top H\big)_\ell=\sum_{x} A_\ell(x)^\top H_x B_\ell(x)^\top .
\end{equation}
$M\dW$ is the first-order change of the operator family induced by $\dW$ at fixed gates. Both
$M$ and $M^\top$ cost $L$ batched matrix products, so neither is ever formed explicitly.

\paragraph{N5. Target, reachable set, projector.} With step size $\eta>0$ the \textbf{target} is
\[
 \Dstar:=\big(-\eta\,G(x)\big)_{x\in\mathcal B}\in\mathcal O ,
\]
the operator step we would take if the operator were a free variable. The \textbf{reachable set}
is $\mathrm{range}(M)\subseteq\mathcal O$ and $\Pi$ denotes orthogonal projection onto it. The
\textbf{achievable ideal} is $\Pi\Dstar$: the closest point to $\Dstar$ that any weight
perturbation can produce at first order.

\paragraph{N6. Comparison of \eqref{eq:coordgrad} with \eqref{eq:Mdef}.} Gradient descent takes
$\dW^{\mathrm{GD}}_\ell=-\mathrm{lr}\,\partial\mathcal L/\partial W_\ell$, and by
\eqref{eq:coordgrad} and \eqref{eq:Mdef} this is exactly $M^\top$ applied to the target (up to the
scalar $\mathrm{lr}/\eta$). Hence
\begin{equation}\label{eq:landw}
 \text{GD's induced operator change}=M M^\top\Dstar,
 \qquad
 \text{achievable ideal}=\Pi\Dstar=M(M^\top M)^{+}M^\top\Dstar .
\end{equation}

\begin{definition}[the three quantities the diagnostic reports]\label{def:diag}
\begin{enumerate}[label=(D\arabic*),leftmargin=*,itemsep=0pt]
  \item \textbf{Reachability} $\alpha:=\|\Pi\Dstar\|/\|\Dstar\|\in[0,1]$ --- how much of the ideal
        the architecture admits at all. Calibration quantity, not a finding.
  \item \textbf{Efficiency} $\epsilon:=\cos\big(MM^\top\Dstar,\ \Pi\Dstar\big)$ --- how much of the
        \emph{achievable} step gradient descent takes. This is the bias, correctly normalised.
  \item \textbf{Spectral effect} --- the effect of the difference on $\sigma(\Jop(x))$: mode
        ordering, effective rank, and the exponent $d\log|\dot s|/d\log s$.
\end{enumerate}
\NOTE{Our earlier grid reported $\cos(\Delta \Jop,G)$, i.e.\ (D2) \emph{without} the projection,
which conflates (D1) and (D2) and forces $\cos\approx0$ whenever $\alpha$ is small. Must be
recomputed.}
\end{definition}

\section{The solvers}\label{sec:solver}

\NOTE{Two different solvers are implemented. They are not interchangeable and the difference
matters; this was previously left unstated.}

\paragraph{S1 --- exact minimal realisation (single target operator; DLN, FGLN).}
Solves
\begin{equation}\label{eq:P}
\min_{\dW}\sum_\ell\|\dW_\ell\|_F^2 \quad\text{s.t.}\quad \prod_\ell(W_\ell+\dW_\ell)=P_\star .
\end{equation}
\emph{Algorithm} (\texttt{studies/scaling.py:dichotomy}): split the chain at its midpoint into
sub-products $Y$ (upper, $b$ layers) and $X$ (lower, $a$ layers); solve the two-factor problem
$\min\|Y'-Y\|^2+\|X'-X\|^2$ s.t. $Y'X'=Z$, started from the closed-form SVD split
$Y_0=US^t,\ X_0=S^{1-t}V^\top$ with $t=a/(a+b)$; recurse into each half. At a degenerate base
point the closed-form split \emph{is} the answer and no iteration runs.
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \OK{} Exact from a degenerate base point for every $L$ with $t=a/(a+b)$ (Thm~\ref{thm:exact});
        verified to $1.000000$, $L=2..128$, $d=32$, ${<}0.05$\,s, residual $2.3\times10^{-14}$.
  \item \OK{} Handles $W=0$, where GD, Adam, Newton, Gauss--Newton, natural gradient and a
        first-order solver on \eqref{eq:P} all return exactly $0$ (Thm~\ref{thm:degen}).
  \item \OK{} Handles ill-conditioning: at Xavier depth 128 the target is reachable to
        $1.7\times10^{-4}$.
  \item \TD{} Away from a degenerate base point it is a heuristic: $\rho=0.05$ ($d{=}32,L{=}8$)
        to $0.23$ ($L{=}32$), budget-saturated. A closed-form cost-to-go exists only under
        spectral separation (ratio $1.18$--$1.36$).
\end{itemize}

\paragraph{S2 --- truncated Krylov iteration on the step map (ReLU, CReLU).}
With input-dependent gates there is no single target operator, so \eqref{eq:P} has no direct
analogue: one $\dW$ must serve $n$ different per-input targets. We therefore solve the linear
least-squares problem in $\mathcal W$
\begin{equation}\label{eq:Q}
  \min_{\dW\in\mathcal W}\ \big\|M\dW-\Dstar\big\|^2
  \qquad\big(\text{or }\ \big\|M\dW-\Dstar\big\|^2+\lambda^2\|\dW\|^2\ \text{ if damped}\big),
\end{equation}
with $M$, $\Dstar$ as in N4--N5. Its minimum-norm solution is $M^{+}\Dstar$ and its image is
exactly the achievable ideal $\Pi\Dstar$.

\emph{Algorithm.} LSQR \cite{paige1982lsqr} applied matrix-free to $M$ (never to $M^\top M$,
which would square the condition number). After $k$ iterations LSQR returns
\[
  \dW^{(k)}=\arg\min_{\dW\in\mathcal K_k}\|M\dW-\Dstar\|,
  \qquad
  \mathcal K_k=\mathrm{span}\Big\{M^\top\Dstar,\ (M^\top M)M^\top\Dstar,\ \dots,\ (M^\top M)^{k-1}M^\top\Dstar\Big\},
\]
the order-$k$ Krylov subspace generated by $M^\top M$ from $M^\top\Dstar$. Two endpoints, both
immediate from this description:
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item $k=1$: $\mathcal K_1=\mathrm{span}\{M^\top\Dstar\}$, which by N6 is the gradient
        direction. So $\dW^{(1)}$ is the gradient step with the residual-minimising step length.
        \RUN{} Measured on MNIST-1D: $\cos(\dW^{(1)},-\nabla\mathcal L)=+1.0000$.
  \item $k\to\infty$: $\dW^{(k)}\to M^{+}\Dstar$, so $M\dW^{(k)}\to\Pi\Dstar$, the achievable ideal.
\end{itemize}
Therefore \textbf{$k$ is an exactly calibrated dial between gradient descent and the best
achievable operator step}, and it is the natural one: truncating a Krylov iteration is a standard
regularisation for ill-posed least squares, damping the small singular directions of $M$ first.

\emph{Cost and conditioning.} One iteration is one application of $M$ and one of $M^\top$, i.e.\
$2L$ batched matrix products. The number of iterations needed to approach $\Pi\Dstar$ is governed
by the singular-value distribution of $M$ restricted to its range; for condition number $\kappa$
the residual decreases roughly like $\big(\frac{\kappa-1}{\kappa+1}\big)^k$.

\emph{Robustness.}
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \OK{} \textbf{Ill-conditioned $M$: handled.} This is what LSQR is for. Truncation at
        moderate $k$, or damping $\lambda$, keeps $\|\dW\|$ bounded; undamped it is not
        ($\|\dW\|\approx12.7$ against a unit-scale target at $\lambda=0$, versus $0.57$ at
        $\lambda=1$). The method trains from Xavier initialisation, which is the ill-conditioned
        case of interest.
  \item \NO{} \textbf{Degenerate $W=0$: fails.} Then $M\equiv0$, so $\mathcal K_k=\{0\}$ and
        $\dW^{(k)}=0$ for every $k$ --- the case Thm~\ref{thm:degen} describes and S1 was built
        for.
  \item \TD{} \textbf{What robustness actually requires.} The exactly-degenerate point is a
        measure-zero corner; the case that matters in practice is a \emph{very deep network at a
        standard initialisation}, where the operator has collapsed but no layer is zero, so $M$ is
        severely ill-conditioned rather than null. There S2 is well posed and truncation is the
        right regulariser. An earlier draft proposed detecting degeneracy and switching to an
        aggregate target; that is wrong twice over --- it assumes all gates coincide, which is
        false when the layers are nonzero, and it requires knowing in advance which regime one is
        in.
  \item \TD{} \textbf{Generic proposal, untested.} Enrich the search space rather than
        special-case it: augment $\mathcal K_k$ with a constructive direction (the balanced-chain
        perturbation for the aggregate target), and choose the coefficients by the \emph{true}
        operator change $\|\prod(W_\ell+\dW_\ell)-\ \cdot\ \|$ evaluated by a forward pass, not by
        the linearised residual. When $M$ is well conditioned the extra direction gets weight
        $\approx0$ and nothing changes; when $M$ degenerates it is the only direction that moves.
        No detection, no exception. This is subspace enrichment in the sense of AMEn.
\end{itemize}

\paragraph{The gap.} \TD{} S1 solves an exact nonlinear constraint for a single target; S2 solves
a linearised problem for a family of targets. There is no exact-constraint analogue of S1 for the
per-input family, because the dichotomy splits \emph{one} target and per-input intermediate
targets conflict. The hybrid above closes the degeneracy hole but not this one.

\section{Supporting results}\label{sec:results}

\begin{proposition}[GD is one Landweber step]\label{prop:landweber}\OK{}\RUN{}
The coordinate gradient is $M^\top\Dstar$, so GD's induced operator step is $MM^\top\Dstar$, while
the best achievable step is $\Pi\Dstar=M(M^\top M)^+M^\top\Dstar$. The gap is the conditioning of
$MM^\top$ on its range.
\end{proposition}

\begin{theorem}[degeneracy]\label{thm:degen}\OK{}\RUN{}
At $W=0$ the product map is homogeneous of degree $L$; all derivatives of order $<L$ vanish, so
every null-consistent derivative-based rule of order $p<L$ returns $\dW=0$.
\end{theorem}

\begin{theorem}[exact recursion]\label{thm:exact}\OK{}\RUN{}
From a degenerate base point, S1 solves \eqref{eq:P} exactly for every $L$ iff each split uses
$t=a/(a+b)$; the symmetric split $t=1/2$ is exact iff $L$ is a power of two.
\end{theorem}

\begin{theorem}[conservation]\label{thm:conserve}\OK{}\RUN{}
Any flow $\dot W_\ell=A_\ell^\top\Lam B_\ell^\top$ with a shared $\Lam$ conserves
$W_{\ell+1}^\top W_{\ell+1}-W_\ell W_\ell^\top$. Hence balancedness
\cite{du2018algorithmic,arora2018optimization} is a property of minimal realisations, not of
gradient descent specifically.
\end{theorem}

\NOTE{Also proved: FGLN gradient flow is deep-linear flow in the variables
$M_\ell=D_\ell W_\ell D_{\ell-1}$, so \cite{arora2018optimization} Thm~1 transfers to FGLN under
$M$-balancedness and \cite{haas2026why} Prop.~4.5 is correct. An earlier claim to the contrary is
withdrawn.}

\section{Where a contribution could come from}\label{sec:gap}

\NOTE{Ordered by how likely they are to survive the literature.}
\begin{enumerate}[leftmargin=*,itemsep=0pt]
  \item \textbf{Finite-time operator dynamics with input-dependent gates} --- the gap named in
        \S\ref{sec:sota}. Concretely: does the $2-2/L$ law hold per-input in a ReLU net, and does
        efficiency (D2) explain the deviation? E2 and E3 are designed for this.
  \item \textbf{Which inputs the step serves.} One update must serve $n$ conflicting per-input
        targets; the allocation is measurable and has no analogue in the linear theory.
  \item \textbf{Conservation (Thm~\ref{thm:conserve})} --- balancedness is a property of minimal
        realisations, not of GD.
  \item \textbf{The solvers} --- tools, reported as such.
\end{enumerate}

\paragraph{Explicitly not claimed.} Gate-induced rank deficiency (self-evident); any new
characterisation of the DLN bias (\cite{arora2018optimization}); a missing FGLN closed form
(withdrawn); a benchmark win (the method realises ${\le}80\%$ of its target at $10\times$ the
wall clock).
