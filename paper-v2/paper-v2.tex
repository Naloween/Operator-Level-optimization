\section{Question, and why this framing}

\textbf{Question.} What is the implicit bias of gradient descent on a factored model, measured
\emph{at the level of the operator the model computes}, and how does the answer change when the
model becomes nonlinear?

\textbf{Framing.} A depth-$L$ network computes a linear operator per input,
$f(x)=\Jop(x)x$. Gradient descent updates \emph{weights}; the operator changes as a consequence.
We take the \textbf{operator gradient step} $-\eta G(x)$, $G=\partial\ell/\partial \Jop$, as the
reference against which the realised change $\Delta \Jop(x)$ is measured. The bias is the
discrepancy. This framing is useful because the reference is architecture-independent: the same
target is defined for a linear network, a fixed-gate network, and a ReLU network, so the
\emph{same} diagnostic runs across all three and its readings are comparable.

\textbf{Programme.} (i) Validate in the deep linear case, where the answer is known in closed
form, and require agreement. (ii) Extend to fixed gates, where the closed form still exists.
(iii) Extend to input-dependent gates (ReLU, CReLU), where it does not, and report what the
diagnostic finds. The contribution is what (iii) shows, not the machinery.

\textbf{What is a contribution and what is not.} The solver and the certificate are tools. That
ReLU gates reduce rank and make the target unreachable is self-evident and is used only to
\emph{calibrate} the diagnostic. The contribution must be empirical or theoretical findings about
the nonlinear regime that are not implied by the deep linear literature.

\section{Definitions}

\begin{definition}[model classes]\label{def:models}
Weights $W_\ell\in\R^{n_\ell\times n_{\ell-1}}$, $\ell=1..L$.
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \textbf{DLN} (deep linear): $\Jop = W_L\cdots W_1$, input-independent.
  \item \textbf{FGLN} (fixed gates): $\Jop = W_LD_{L-1}\cdots D_1W_1$, $D_\ell$ fixed diagonal.
  \item \textbf{ReLU MLP} (no biases): $\Jop(x)=W_LD_{L-1}(x)\cdots D_1(x)W_1$ with
        $D_\ell(x)=\mathrm{diag}(\mathbb 1[z_\ell(x)>0])$, so $f(x)=\Jop(x)x$ exactly.
  \item \textbf{CReLU MLP}: $c=[\mathrm{relu}(z);\mathrm{relu}(-z)]$, $W\in\R^{d\times 2d}$;
        $D(z)=[\mathrm{diag}(\mathbb 1[z>0]);-\mathrm{diag}(\mathbb 1[z<0])]$ satisfies
        $D^\top D=I$, so the gate is an isometry and looks-linear init gives $\Jop(x)=O_L\cdots O_1$
        orthogonal at any depth.
\end{itemize}
\end{definition}

\begin{definition}[contexts, target, realised step]\label{def:contexts}
$\Jop(x)=A_\ell(x)W_\ell B_\ell(x)$ with $B_1=I$, $A_L=I$,
$B_{\ell+1}=D_\ell W_\ell B_\ell$, $A_\ell=A_{\ell+1}W_{\ell+1}D_\ell$.
Coordinate gradient $\partial\mathcal L/\partial W_\ell=\sum_x A_\ell(x)^\top G(x)B_\ell(x)^\top$.
Target $\Dstar=(-\eta G(x))_x$. Reachable set at first order:
$M(\dW)=\big(\sum_\ell A_\ell(x)\dW_\ell B_\ell(x)\big)_x$, and $\Pi$ its orthogonal projector.
\end{definition}

\begin{definition}[the three quantities the diagnostic reports]\label{def:diag}
\begin{enumerate}[label=(D\arabic*),leftmargin=*,itemsep=0pt]
  \item \textbf{Reachability} $\ \alpha=\|\Pi\Dstar\|/\|\Dstar\|$ --- how much of the ideal the
        architecture admits at all. Calibration quantity, not a finding.
  \item \textbf{Efficiency} $\ \epsilon=\cos(\Delta \Jop_{\mathrm{GD}},\,\Pi\Dstar)$ --- how much
        of the \emph{achievable} step gradient descent takes. This is the bias, correctly
        normalised.
  \item \textbf{Spectral effect} --- what the difference does to $\sigma(\Jop)$: mode ordering,
        effective rank, and the separation exponent $d\log|\dot s|/d\log s$.
\end{enumerate}
\NOTE{Our earlier grid reported $\cos(\Delta \Jop,G)$, i.e.\ (D2) without the projection. That
conflates (D1) and (D2) and forces $\cos\approx0$ whenever $\alpha$ is small. Must be recomputed.}
\end{definition}

\section{The solvers}\label{sec:solver}

\NOTE{Two different solvers are implemented. They are not interchangeable and the difference
matters; this was previously left unstated.}

\paragraph{S1 --- exact minimal realisation (single target operator; DLN, FGLN).}
Solves
\begin{equation}\label{eq:P}
\min_{\dW}\sum_\ell\|\dW_\ell\|_F^2 \quad\text{s.t.}\quad \prod_\ell(W_\ell+\dW_\ell)=P_\star .
\end{equation}
\emph{Algorithm} (\texttt{studies/scaling.py:dichotomy}): split the chain at its midpoint into
sub-products $Y$ (upper, $b$ layers) and $X$ (lower, $a$ layers); solve the two-factor problem
$\min\|Y'-Y\|^2+\|X'-X\|^2$ s.t. $Y'X'=Z$, started from the closed-form SVD split
$Y_0=US^t,\ X_0=S^{1-t}V^\top$ with $t=a/(a+b)$; recurse into each half. At a degenerate base
point the closed-form split \emph{is} the answer and no iteration runs.
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \OK{} Exact from a degenerate base point for every $L$ with $t=a/(a+b)$ (Thm~\ref{thm:exact});
        verified to $1.000000$, $L=2..128$, $d=32$, ${<}0.05$\,s, residual $2.3\times10^{-14}$.
  \item \OK{} Handles $W=0$, where GD, Adam, Newton, Gauss--Newton, natural gradient and a
        first-order solver on \eqref{eq:P} all return exactly $0$ (Thm~\ref{thm:degen}).
  \item \OK{} Handles ill-conditioning: at Xavier depth 128 the target is reachable to
        $1.7\times10^{-4}$.
  \item \TD{} Away from a degenerate base point it is a heuristic: $\rho=0.05$ ($d{=}32,L{=}8$)
        to $0.23$ ($L{=}32$), budget-saturated. A closed-form cost-to-go exists only under
        spectral separation (ratio $1.18$--$1.36$).
\end{itemize}

\paragraph{S2 --- per-input least squares (ReLU, CReLU).}
With input-dependent gates there is no single target operator, so \eqref{eq:P} has no direct
analogue. \texttt{studies/dial.py} instead solves the first-order problem
\begin{equation}\label{eq:Q}
\min_{\dW}\ \big\|M(\dW)-\Dstar\big\|^2 \quad\text{by LSQR truncated at $k$ iterations.}
\end{equation}
\begin{itemize}[leftmargin=*,itemsep=0pt]
  \item \OK{} $k$ is an exactly calibrated dial: $k{=}1$ gives $\cos(\dW,-\nabla)=+1.0000$
        (Prop.~\ref{prop:landweber}), $k\to\infty$ gives the best achievable step.
  \item \NO{} \textbf{Linearised, hence pinned at $W=0$} --- precisely the case S1 was built to
        handle. \textbf{S2 is not robust in the regime we need robustness for.}
  \item \TD{} No exact-constraint analogue of S1 for the per-input family: the dichotomy splits
        \emph{one} target, and per-input intermediate targets conflict. Unsolved.
\end{itemize}

\paragraph{The gap.} \TD{} A single solver valid across all four model classes, degenerate and
ill-conditioned included. Without it the diagnostic's readings in the nonlinear regime rest on a
method that would fail the robustness test we set ourselves.

\section{Supporting results}\label{sec:results}

\begin{proposition}[GD is one Landweber step]\label{prop:landweber}\OK{}\RUN{}
The coordinate gradient is $M^\top\Dstar$, so GD's induced operator step is $MM^\top\Dstar$, while
the best achievable step is $\Pi\Dstar=M(M^\top M)^+M^\top\Dstar$. The gap is the conditioning of
$MM^\top$ on its range.
\end{proposition}

\begin{theorem}[degeneracy]\label{thm:degen}\OK{}\RUN{}
At $W=0$ the product map is homogeneous of degree $L$; all derivatives of order $<L$ vanish, so
every null-consistent derivative-based rule of order $p<L$ returns $\dW=0$.
\end{theorem}

\begin{theorem}[exact recursion]\label{thm:exact}\OK{}\RUN{}
From a degenerate base point, S1 solves \eqref{eq:P} exactly for every $L$ iff each split uses
$t=a/(a+b)$; the symmetric split $t=1/2$ is exact iff $L$ is a power of two.
\end{theorem}

\begin{theorem}[conservation]\label{thm:conserve}\OK{}\RUN{}
Any flow $\dot W_\ell=A_\ell^\top\Lam B_\ell^\top$ with a shared $\Lam$ conserves
$W_{\ell+1}^\top W_{\ell+1}-W_\ell W_\ell^\top$. Hence balancedness
\cite{du2018algorithmic,arora2018optimization} is a property of minimal realisations, not of
gradient descent specifically.
\end{theorem}

\NOTE{Also proved: FGLN gradient flow is deep-linear flow in the variables
$M_\ell=D_\ell W_\ell D_{\ell-1}$, so \cite{arora2018optimization} Thm~1 transfers to FGLN under
$M$-balancedness and \cite{haas2026why} Prop.~4.5 is correct. An earlier claim to the contrary is
withdrawn.}

The diagnostic must give correct readings where ordinary training fails: DLN at Xavier depth 128,
ReLU at a depth where Adam collapses, and $W=0$. \textbf{Blocked on the S1/S2 gap

\paragraph{E5 --- Real data. \RUN{} partial}

\section{Where a contribution could come from}\label{sec:gap}

\NOTE{Ordered by how likely they are to survive the literature.}
\begin{enumerate}[leftmargin=*,itemsep=0pt]
  \item \textbf{Finite-time operator dynamics with input-dependent gates} --- the gap named in
        \S\ref{sec:sota}. Concretely: does the $2-2/L$ law hold per-input in a ReLU net, and does
        efficiency (D2) explain the deviation? E2 and E3 are designed for this.
  \item \textbf{Which inputs the step serves.} One update must serve $n$ conflicting per-input
        targets; the allocation is measurable and has no analogue in the linear theory.
  \item \textbf{Conservation (Thm~\ref{thm:conserve})} --- balancedness is a property of minimal
        realisations, not of GD.
  \item \textbf{The solvers} --- tools, reported as such.
\end{enumerate}

\paragraph{Explicitly not claimed.} Gate-induced rank deficiency (self-evident); any new
characterisation of the DLN bias (\cite{arora2018optimization}); a missing FGLN closed form
(withdrawn); a benchmark win (the method realises ${\le}80\%$ of its target at $10\times$ the
wall clock).
